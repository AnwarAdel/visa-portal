    let isPreviewActive = false;
    let globalLogoBase64 = "";

    function loadCustomLogo(event) {
      const file = event.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(e) {
          globalLogoBase64 = e.target.result;
          localStorage.setItem('savedCustomLogo', globalLogoBase64);
          applyLogoToTemplates(globalLogoBase64);
        };
        reader.readAsDataURL(file);
      }
    }

    function applyLogoToTemplates(logoSrc) {
      const imgT2 = document.getElementById('displayLogoT2');
      const imgT3 = document.getElementById('displayLogoT3');
      if (logoSrc) {
        imgT2.src = logoSrc;
        imgT2.style.display = 'block';
        imgT3.src = logoSrc;
        imgT3.style.display = 'block';
      }
    }

    function formatDate(dateStr) {
      if (!dateStr) return '';
      return dateStr.replace(/-/g, '/');
    }

    function generateAll() {
      const name = document.getElementById('p_name').value;
      const nat = document.getElementById('p_nat').value;
      const natEn = document.getElementById('p_nat_en').value;
      const pass = document.getElementById('p_passport').value;
      const dob = formatDate(document.getElementById('p_dob').value);
      const airline = document.getElementById('p_airline').value;
      const airlineEn = document.getElementById('p_airline_en').value;
      const flight = document.getElementById('p_flight').value;
      const dest = document.getElementById('p_dest').value;
      const from = document.getElementById('p_from').value;
      const issueDate = formatDate(document.getElementById('p_issue_date').value);
      const expiryDate = formatDate(document.getElementById('p_expiry_date').value);
      const flightDate = formatDate(document.getElementById('p_flight_date').value);
      const pnr = document.getElementById('p_pnr').value;
      const appNo = document.getElementById('p_app_no').value;
      const docRef = document.getElementById('p_doc_ref').value;

      const now = new Date();
      const options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric', hour: '2-digit', minute: '2-digit', hour12: true };
      const realDateTimeStr = now.toLocaleDateString('en-US', options);
      document.getElementById('out_real_datetime').innerText = realDateTimeStr;

      const passengerData = { name, nat, natEn, pass, dob, airline, airlineEn, flight, dest, flightDate, appNo, docRef, realDateTimeStr };
      localStorage.setItem('latestPassenger', JSON.stringify(passengerData));

      // 1. تعبئة القالب الأول
      document.getElementById('out_issue').innerText = issueDate;
      document.getElementById('out_expiry').innerText = expiryDate;
      document.getElementById('out_date1').innerText = flightDate;
      document.getElementById('out_name1').innerText = name;
      document.getElementById('out_nat1').innerText = nat;
      document.getElementById('out_pass1').innerText = pass;
      document.getElementById('out_dest1').innerText = dest;
      document.getElementById('out_from1').innerText = from;
      document.getElementById('out_flight1').innerText = flight;
      document.getElementById('out_pnr1').innerText = pnr;
      document.getElementById('out_pnr2').innerText = pnr;
      document.getElementById('out_airline1').innerText = airline;

      // 2. تعبئة القالب الثاني
      document.getElementById('out_app_no').innerText = appNo;
      document.getElementById('out_app_date').innerText = issueDate;
      document.getElementById('out_doc_ref').innerText = `Document Ref: ${docRef}`;
      document.getElementById('out_name2').innerText = name;
      document.getElementById('out_pass2').innerText = pass;
      document.getElementById('out_dob2').innerText = dob;
      document.getElementById('out_nat2').innerText = nat;
      document.getElementById('out_nat_en2').innerText = natEn;
      document.getElementById('out_airline2').innerText = airline;
      document.getElementById('out_airline_en2').innerText = airlineEn;
      document.getElementById('out_flight2').innerText = flight;
      document.getElementById('out_date2').innerText = flightDate;

      // 3. تعبئة القالب الثالث
      updateTemplate3(passengerData);

      // ربط الـ QR بشرط فتح صفحة التحقق (القالب الثالث) مباشرة بدلاً من السيرفر الخارجي
      const currentFullUrl = window.location.href.split('?')[0];
      const encodedVerifyUrl = `${currentFullUrl}?mode=verify&pass=${encodeURIComponent(pass)}&name=${encodeURIComponent(name)}&nat=${encodeURIComponent(nat)}&airline=${encodeURIComponent(airline)}&flight=${encodeURIComponent(flight)}&dest=${encodeURIComponent(dest)}&date=${encodeURIComponent(flightDate)}`;

      document.getElementById("qrcode").innerHTML = "";
      new QRCode(document.getElementById("qrcode"), {
        text: encodedVerifyUrl,
        width: 105,
        height: 105
      });
    }

    function togglePreviewMode() {
      isPreviewActive = !isPreviewActive;
      const formPanel = document.getElementById('formPanel');
      const previewBanner = document.getElementById('previewBanner');
      const exitBtn = document.getElementById('exitPreviewBtn');

      if (isPreviewActive) {
        formPanel.style.display = 'none';
        previewBanner.style.display = 'block';
        exitBtn.style.display = 'block';
        window.scrollTo({ top: 0, behavior: 'smooth' });
      } else {
        formPanel.style.display = 'block';
        previewBanner.style.display = 'none';
        exitBtn.style.display = 'none';
      }
    }

      document.getElementById('template3View').style.display = 'none';
    }

    window.onload = function() {
      const savedLogo = localStorage.getItem('savedCustomLogo');
      if (savedLogo) {
        applyLogoToTemplates(savedLogo);
      }

      const urlParams = new URLSearchParams(window.location.search);
      if (urlParams.get('mode') === 'verify') {
        // عند مسح الـ QR من كاميرا الهاتف
        document.getElementById('formPanel').style.display = 'none';
        document.getElementById('mainTemplatesWrapper').style.display = 'none';
        
        const t3 = document.getElementById('template3View');
        t3.style.display = 'block';
        t3.style.margin = '40px auto';

        // استخدام البيانات الآتية في الرابط مباشرة
        const verifyData = {
          name: urlParams.get('name'),
          pass: urlParams.get('pass'),
          nat: urlParams.get('nat'),
          airline: urlParams.get('airline'),
          flight: urlParams.get('flight'),
          dest: urlParams.get('dest'),
          flightDate: urlParams.get('date')
        };

        updateTemplate3(verifyData);
      } else {
        generateAll();
      }
